\begin{lemma}[Actual-source support]
\label{lem:support}
Let $T=4$, $\varepsilon=\tfrac12$, $U=(0.9,1.1)\times(-0.1,0.1)^3$, and let
$\chi$ be the XII-D ramp profile. Define
\begin{align*}
S(u)&=b(2u-5),\qquad
b(v)=e^{1-1/(1-v^2)}\ (|v|<1),\ 0\ \text{else},\\
\beta(t)&=\chi\bigl((t-1.8)/0.1\bigr)\chi\bigl((3.2-t)/0.1\bigr),\\
\lambda(r)&=\chi\bigl((r-0.6)/0.1\bigr)\chi\bigl((2.5-r)/0.2\bigr),\\
\kappa_0(t)&=\chi\bigl((t-0.55)/0.05\bigr)\chi\bigl((3.6-t)/0.1\bigr),\\
\mu(r)&=\chi\bigl((3.2-r)/0.2\bigr),
\end{align*}
so $S\in C_c^\infty(\mathbb{R};[0,1])$ with $\operatorname{supp}S=[2,3]$ (not claimed
inside $(2,3)$),
$\operatorname{supp}\beta=[1.8,3.2]$ with $\beta=1$ on $[1.9,3.1]$,
$\operatorname{supp}\lambda=[0.6,2.5]$ with $\lambda=1$ on $[0.7,2.3]$,
$\kappa_0=1$ on $[0.6,3.5]\supset\operatorname{supp}\beta$, and
$\mu=1$ on $\{r\le3\}\supset\operatorname{supp}\lambda$.
Set $\psi=\beta(t)\lambda(r)r^2$, $f=\square\psi$, $v_f=E_0^{\mathrm A}f=\psi$,
$g=\kappa_0(t)\mu(r)(1-S(t))$, and $\sigeps(t)=\chi(t/\varepsilon)$.
Then:
\begin{enumerate}
\item $g\in C_c^\infty(M_T;[0,1])$, $g=\sigma=1$ on a neighbourhood of $\overline U$,
  and $\overline U\subset\operatorname{Int}\{g=1\}\cap\{\varepsilon<t<T\}$.
\item \textbf{(Global support.)}
  On a neighbourhood of $\operatorname{supp}d\psi$ one has $\kappa_0=\mu=\sigma=1$, hence
  $g-\sigma=-S$. Outside $\operatorname{supp}d\psi$, the massless special-family identity
  $\alpha=(g-\sigma)\,\mathfrak a^{\mu1}\partial_\mu v_f$ (XII-D eq:alpha at $m=0$,
  $\phi=x^1$, $v_f=\psi$) forces $\alpha=0$. Therefore
  \[
  \alpha=-S(t)\,\mathfrak a^{\mu1}\partial_\mu\psi
  \qquad\text{globally on }M_T,
  \]
  and $\operatorname{supp}_t(D_{\mathrm{div}}\alpha)\subset[2,3]$.
\item On the open set $N=\{t\in(1.9,3.1)\}\times\{0.7<r<2.3\}$ one has
  $\beta=\lambda=\kappa_0=\mu=\sigma=1$ with all derivatives vanishing, hence
  $\psi=r^2$ and
  \[
  D_{\mathrm{div}}\alpha=(S'x,\,3S,\,0,\,0)
  \]
  on $N$. Combined with item~2, $D_{\mathrm{div}}\alpha$ vanishes off the time slab
  $[2,3]$: temporal cutoff edges of $\beta$ and the $\kappa_0/\mu$ transitions
  contribute no source; radial cutoff contributions of $\lambda$ during $[2,3]$
  do not meet the relevant advanced cones from $\overline U$ (item~4).
  (No claim is made that radial edges contribute no source globally.)
\item Every advanced cone from a point of $\overline U$ meets
  $\{t\in\operatorname{supp}S\}$ only at radii in $[2-1.1,\,3-0.9]=[0.9,2.1]$,
  hence at spatial radii in $[0.726,2.274]\subset(0.7,2.3)$. Those cone points
  therefore lie in $N$. By items~2--3 the future cones from $U$ see exactly the
  model source $D_{\mathrm{div}}\alpha=(S'x,3S,0,0)$ (no exterior-$N$ contribution).
\item The collar $\vartheta$ of XII-D eq:compact-root equals $1$ on a neighbourhood
  of $[0,T]$, hence on $U$ and on every cone point used above; insertion of
  $\vartheta$ changes nothing (XII-D thm:root). The independent switch
  $\chi_{\mathrm I}$ of XVI-J/K does not occur in $t_{\mathrm D}$ (XVI-I eq:push /
  XVI-H thm:trep / XII-D $B_{f,g}$) and does not affect $F_t$.
\item Consequently the local root formulas on $U$,
  \[
  w=(-m_0x,\;3(m_1-m_0t),\;0,\;0),\qquad
  H^{01}=2m_0,
  \]
  follow from the actual compact construction by $4$D Huygens (cone support of
  $E_0^{\mathrm A}$), not from polynomial surrogate sources.
\end{enumerate}
\end{lemma}
\begin{proof}
Item~1 is an exact containment of the support intervals.
For item~2: $\operatorname{supp}d\psi\subset\operatorname{supp}\beta\times\operatorname{supp}\lambda$
up to the radial factor $r^2$, and on that set $\kappa_0=\mu=1$ by construction while
$\sigma=1$ because $t\ge1.8>\varepsilon$; thus $g-\sigma=-S$ on a neighbourhood of
$\operatorname{supp}d\psi$. Off $\operatorname{supp}d\psi$ one has $\partial\psi=0$, so the
special-family factorisation $\alpha=(g-\sigma)\mathfrak a^{\mu1}\partial_\mu\psi$ yields
$\alpha=0$. The global formula and the time support of $D_{\mathrm{div}}\alpha$ follow.
Item~3 is the symbolic reduction of XII-D eq:alpha on the plateau set $N$
(\texttt{nins3\_support\_lemma\_certificates.json}, Parts~P1--P2), together with item~2
excluding all source outside the time slab $[2,3]$.
Item~4 is the cone-radius computation
$R\in[2-1.1,\,3-0.9]$ and
$r\in[R-\sqrt3\cdot0.1,\,R+\sqrt3\cdot0.1]\subset(0.7,2.3)$, so every contributing
cone point lies in $N$ where the model source holds.
Item~5 quotes XII-D thm:root and the XVI-K inheritance recorded in
Section~\ref{sec:tD}. Item~6 is then the Kirchhoff evaluation of
Proposition~\ref{prop:4Dsf}. Numeric corroboration (Part~P3 of the certificate)
reproduces $J_{\mathrm{actual}}=J_{\mathrm{model}}$ at random cone points to
$3\times10^{-21}$ and the closed forms for $w$ on $U$ to relative
$10^{-14}$; it is corroboration only.
\end{proof}
